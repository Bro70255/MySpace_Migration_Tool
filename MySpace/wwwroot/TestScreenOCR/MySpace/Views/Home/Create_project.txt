<script>
    let flow = [];

    function addFlowStepAuto() {
        const select = document.getElementById("flowStep");
        const step = select.value;

        if (!step) return;

        flow.push(step);
        renderFlow();
        updateHiddenField();
        select.value = "";
    }

    function removeStep(index) {
        flow.splice(index, 1);
        renderFlow();
        updateHiddenField();
    }

    function renderFlow() {
        const container = document.getElementById("flowPipeline");
        container.innerHTML = "";

        flow.forEach((step, index) => {
            const node = document.createElement("div");
            node.className = "flow-node";
            node.innerHTML = `
                <span>${step}</span>
                <span class="flow-remove" onclick="removeStep(${index})">✕</span>
            `;

            container.appendChild(node);

            if (index < flow.length - 1) {
                const arrow = document.createElement("div");
                arrow.className = "flow-arrow";
                arrow.innerHTML = "→";
                container.appendChild(arrow);
            }
        });
    }

    function updateHiddenField() {
        document.getElementById("ProjectFlow").value = JSON.stringify(flow);
    }
</script>
