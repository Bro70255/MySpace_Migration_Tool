<script>
    $(document).ready(function () {

        // Expand / Collapse All checkbox
        document.addEventListener("change", function (e) {
            if (e.target.id === "toggleAll") {
                const expand = e.target.checked;

                document.querySelectorAll(".tree-children").forEach(ul => {
                    ul.classList.toggle("open", expand);
                });

                document.querySelectorAll(".tree-caret").forEach(caret => {
                    caret.classList.toggle("open", expand);
                });

                document.querySelectorAll(".tree-folder-icon").forEach(icon => {
                    icon.textContent = expand ? "📂" : "📁";
                });
            }
        });

        loadOCRTreeView();
    });

</script>
