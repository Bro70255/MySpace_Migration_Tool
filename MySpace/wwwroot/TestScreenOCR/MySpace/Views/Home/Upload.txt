<div class="content-wrapper">

    <div class="glass-body module-wrapper">

        <!-- ================= HEADER ================= -->
        <div class="module-header">
            <span class="module-icon">📤</span>
            <span class="module-title">Upload Screens</span>
        </div>

        <div class="module-body">

            <!-- ================= MESSAGE ================= -->
            <div id="uploadMessage" class="upload-message" style="display:none;"></div>

            <!-- ================= PROJECT + FILE TYPE ================= -->
            <div class="project-select-wrapper">

                <div class="project-row">
                    <div class="project-field">
                        <label class="project-select-label">Project</label>
                        <select id="projectSelect" class="project-select">
                            <option value="">-- Select Project --</option>
                        </select>
                    </div>

                    <div class="project-field">
                        <label class="project-select-label">File Type</label>
                        <select id="fileTypeSelect" class="project-select">
                            <option value="">-- Select File Type --</option>
                        </select>
                    </div>
                </div>

                <div id="projectHint" class="project-select-hint">
                    Please select a project before uploading
                </div>
            </div>

            <!-- ================= UPLOAD ZONE ================= -->
            <div id="dropZone" class="upload-zone">

                <p class="upload-title">Upload Screens</p>
                <p class="upload-subtitle">Drag & drop folders or files here</p>
                <span class="upload-separator">or</span>

                <!-- Hidden Inputs -->
                <input type="file" id="folderInput" multiple webkitdirectory hidden />
                <input type="file" id="fileInput" multiple hidden />

                <div class="upload-actions">
                    <button type="button" class="modern-btn" id="btnFolder">
                        Browse Folder
                    </button>

                    <button type="button" class="modern-btn" id="btnFiles">
                        Browse Files
                    </button>
                </div>

                <div class="upload-info" id="uploadInfo"></div>

            </div>

        </div>
    </div>

@section Scripts {
    <script>
        $(document).ready(function () { loadProjects(); });
           let selectedFiles = [];

        function showMessage(message, type) {
            const msg = document.getElementById("uploadMessage");
            msg.className = `upload-message ${type}`;
            msg.innerText = message;
            msg.style.display = "block";
            setTimeout(() => msg.style.display = "none", 4000);
        }

        function validateSelections(projectSelect, fileTypeSelect, projectHint) {
            if (!projectSelect.value) { projectHint.style.display = "block"; showMessage("Please select a project", "error"); return false; }
            if (!fileTypeSelect.value) { projectHint.style.display = "block"; showMessage("Please select a file type", "error"); return false; }
            projectHint.style.display = "none";
            return true;
        }

        function buildFormData(projectSelect, fileTypeSelect, files) {
            const formData = new FormData();
            const projectId = projectSelect.value;
            const projectName = projectSelect.options[projectSelect.selectedIndex]?.text || "";

            formData.append("projectId", projectId);
            formData.append("projectName", projectName);
            formData.append("fileType", fileTypeSelect.value);

            // IMPORTANT: keep relative path when folder upload (webkitdirectory)
            Array.from(files).forEach(f => formData.append("files", f, f.webkitRelativePath || f.name));
            return formData;
        }

        function uploadNow(projectSelect, fileTypeSelect, files) {
            const formData = buildFormData(projectSelect, fileTypeSelect, files);

            // CHANGE URL if your controller route is different
            $.ajax({
                url: "/Home/UploadScreenFolder",
                type: "POST",
                data: formData,
                processData: false,
                contentType: false,
                success: function (res) {
                    if (res && res.success) showMessage(res.message || "Uploaded successfully", "success");
                    else showMessage(res.message || "Upload failed", "error");
                },
                error: function (xhr) {
                    showMessage("Upload error: " + (xhr.responseText || xhr.statusText || "Unknown"), "error");
                }
            });
        }

        document.addEventListener("DOMContentLoaded", () => {
            const dropZone = document.getElementById("dropZone");
            const folderInput = document.getElementById("folderInput");
            const fileInput = document.getElementById("fileInput");
            const btnFolder = document.getElementById("btnFolder");
            const btnFiles = document.getElementById("btnFiles");
            const projectSelect = document.getElementById("projectSelect");
            const fileTypeSelect = document.getElementById("fileTypeSelect");
            const projectHint = document.getElementById("projectHint");
            const uploadInfo = document.getElementById("uploadInfo");

            btnFolder.addEventListener("click", () => { if (!validateSelections(projectSelect, fileTypeSelect, projectHint)) return; folderInput.click(); });
            btnFiles.addEventListener("click", () => { if (!validateSelections(projectSelect, fileTypeSelect, projectHint)) return; fileInput.click(); });

            folderInput.addEventListener("change", e => handleFiles(e.target.files));
            fileInput.addEventListener("change", e => handleFiles(e.target.files));

            dropZone.addEventListener("dragover", e => { e.preventDefault(); dropZone.classList.add("drag-active"); });
            dropZone.addEventListener("dragleave", () => dropZone.classList.remove("drag-active"));
            dropZone.addEventListener("drop", e => { e.preventDefault(); dropZone.classList.remove("drag-active"); if (!validateSelections(projectSelect, fileTypeSelect, projectHint)) return; handleFiles(e.dataTransfer.files); });

            function handleFiles(files) {
                if (!files || files.length === 0) return;

                selectedFiles = files;

                uploadInfo.innerHTML = `<div class="upload-count">${files.length} file(s) selected</div><ul></ul>`;
                const ul = uploadInfo.querySelector("ul");
                Array.from(files).forEach(f => { const li = document.createElement("li"); li.textContent = f.webkitRelativePath || f.name; ul.appendChild(li); });

                // Auto upload after selection (remove this if you want a manual Upload button)
                uploadNow(projectSelect, fileTypeSelect, files);
            }
        });

    </script>
}
