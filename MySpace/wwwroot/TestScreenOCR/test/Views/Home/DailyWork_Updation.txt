<script>
    Bind_Module_fordailywork();
    Bind_CRF_fordailywork();
    Bind_Description_of_work();

    var today = new Date().toISOString().split('T')[0];
    document.getElementById('dwu_date').value = today;
    document.getElementById('dwu_date').addEventListener('input', function () {
        this.value = today;
    });
    $(document).ready(function () {
        $("#loading").hide();
        $("#addButton").click(function () {
            if ($('#tbltable tr').length > 1) { // Check if table has more than 1 row (assuming header row exists)
                $('#tbltable').parent().css('display', 'block');
                $('#DW_btn_Conform').css('display', 'block');
            } else {
                $('#tbltable').parent().css('display', 'none'); // Hide table if no data
                $('#DW_btn_Conform').css('display', 'none'); // Hide button if no data
            }
        });
    });

    document.addEventListener("DOMContentLoaded", function () {
        var crfDropdown = document.getElementById("crf_withsub");
        var crfOptionExists = false;

        // Check if any option has value "crf"
        for (var i = 0; i < crfDropdown.options.length; i++) {
            if (crfDropdown.options[i].value === "crf") {
                crfOptionExists = true;
                break;
            }
        }

        // Remove any existing "Others" options
        var othersOptions = document.querySelectorAll('#crf_withsub option[value="others"]');
        othersOptions.forEach(function (option) {
            option.remove();
        });

        // If "crf" option exists, remove the "Others" option
        if (!crfOptionExists) {
            // If "crf" option doesn't exist, add "Others" option
            var othersOption = document.createElement("option");
            othersOption.value = "others";
            othersOption.text = "Others";
            crfDropdown.appendChild(othersOption);
        }

        // Adding event listener for when the dropdown value changes
        crfDropdown.addEventListener("change", function () {
            if (this.value === "others") {
                // Call the function to save "Others" in the table
                saveOthersToTable();
            }
        });
    });

    function saveOthersToTable() {
        // Assuming you want to get the value from the dropdown and save it to the table
        var selectedText = $("#crf_withsub option:selected").text();
        var newRow = $("<tr>");
        newRow.append("<td>" + selectedText + "</td>");
        $("#tbtable tbody").prepend(newRow);
    }

</script>
