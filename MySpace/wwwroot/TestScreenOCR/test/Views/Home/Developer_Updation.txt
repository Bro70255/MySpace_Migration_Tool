    $(document).ready(function () {
        $("#loading").hide();

            var selectedStatus = $(this).val();
            if (selectedStatus === '12') {
                $('.pop_div').show();
                $('.imp_heading').show();
            } else {
                $('.pop_div').hide();
                $('.imp_heading').hide();
            }
        });
        // Trigger change event on page load if status is initially set to 12
        if ($('#workstatus').val() === '12') {
            $('#workstatus').change();
        }
    });
    function handleSelectChange() {
        var selectElement = document.getElementById('crf_with_sub');
        var selectedValue = selectElement.value;
        All_Attachments(selectedValue);
    }
</script>
