<script>
    initialization_Teach_Lead_Team();
    Bind_Technicalanalysis_Techlead();
    Bind_Developers();
    Bind_Techlead();
    Bind_Change_Type();
    Bind_CRF_Complexity();
    $(document).ready(function () {
        $("#loading").hide();
        $("#dev_add_btn").click(function () {

            var Phase = $("#phase").val();
            if (Phase == "") {
                alert("Please enter the phase");
                return;
            }
            var changes = $("#changes").val();
            if (changes == "") {
                alert("Please enter the no. of changes");
                return; 0
            }
            var start_date = $("#start_date").val();
            if (start_date == "") {
                alert("Please enter the start_date");
                return;
            }
            var end_date = $("#end_date").val();
            if (end_date == "") {
                alert("Please enter the end_date");
                return;
            }
            var hours = $("#hours").val();
            if (hours == "") {
                alert("Please enter the hours");
                return;
            }
            var costestimation = $("#costestimation").val();
            if (costestimation == "") {
                alert("Please enter the costestimation");
                return;
            }

            // Extract developer ID from the selected option
            var DeveloperDropdown = $("#ddldeveloper");
            var DeveloperId = DeveloperDropdown.val();

            if (DeveloperId == "0") {
                alert("Please Select the developer");
                return;
            }
            var Developer = DeveloperDropdown.find("option:selected").text(); // Get the text of the selected option

            // Ensure that DeveloperId is not null and is a non-empty string
            if (!DeveloperId && DeveloperDropdown.find("option:selected").length > 0) {
                DeveloperId = DeveloperDropdown.find("option:selected").val();
            }

            var ChangetypeDropdown = $("#ddlchange_type");
            var ChangetypeId = ChangetypeDropdown.val();
            if (ChangetypeId == "0") {
                alert("Please Select the changes");
                return;
            }
            var Change = ChangetypeDropdown.find("option:selected").text(); // Get the text of the selected option

            var crf_complexity = $("#ddlCRF_Complexity");
            var crf_complexity_id = crf_complexity.val();
            if (crf_complexity_id == "0") {
                alert("Please Select the CRF Complexity");
                return;
            }
            var CRF_complexity = crf_complexity.find("option:selected").text(); // Get the text of the selected option


            if (!ChangetypeId && ChangetypeDropdown.find("option:selected").length > 0) {
                ChangetypeId = ChangetypeDropdown.find("option:selected").val();
            }

            if (!crf_complexity_id && crf_complexity.find("option:selected").length > 0) {
                crf_complexity_id = crf_complexity.find("option:selected").val();
            }

            var RelatedWorkDropdown = $("#ddlrelated_work");
            var RelatedWorkId = RelatedWorkDropdown.val();
            if (RelatedWorkId == "0") {
                alert("Please Select the Related Work");
                return;
            }
            var Work = RelatedWorkDropdown.find("option:selected").text(); // Get the text of the selected option

            // Ensure that DeveloperId is not null and is a non-empty string
            if (!RelatedWorkId && RelatedWorkDropdown.find("option:selected").length > 0) {
                RelatedWorkId = RelatedWorkDropdown.find("option:selected").val();
            }

            var Development = parseFloat($("#development").val()) || 0; // Parse as float, default to 0 if not a number

            var CodeReviewMultiplier = 0.40;
            // Calculate CodeReview value
            var CodeReview = Development * CodeReviewMultiplier;

            var Number = $("#changes").val();
            var Manpower = $("#hours").val();
            var StartDt = $("#start_date").val();
            var EndDt = $("#end_date").val();
            var Cost = $("#costestimation").val();

            if (Phase && ChangetypeId && Change && crf_complexity_id && CRF_complexity && RelatedWorkId && Work && !isNaN(Development) && Manpower && DeveloperId && Developer && StartDt && EndDt && Cost) {
                // Add row to the table with data-developer-id attribute
                var row = $("<tr>" +
                    "<td>" + Phase + "</td>" +
                    "<td style='display: none'>" + ChangetypeId + "</td>" +
                    "<td>" + Change + "</td>" +
                    "<td style='display: none'>" + crf_complexity_id + "</td>" +
                    "<td>" + CRF_complexity + "</td>" +
                    "<td style='display: none'>" + RelatedWorkId + "</td>" +
                    "<td>" + Work + "</td>" +
                    "<td class='number-column'>" + Number + "</td>" +
                    "<td class='development-column'>" + Manpower + "</td>" +
                    "<td style='display: none'>" + DeveloperId + "</td>" +
                    "<td>" + Developer + "</td>" +
                    "<td>" + StartDt + "</td>" +
                    "<td>" + EndDt + "</td>" +
                    "<td class='cost-column'>" + Cost + "</td>" +
                    "<td><button class='btn btn-danger btn-remove'>Remove</button></td>" +
                    "</tr>");

                // Append the developer ID as a hidden input in the last column
                /* row.find("td:last").append("<input type='hidden' name='developer-id' value='" + DeveloperId + "'>");*/

                $("#tbtable1").append(row);

                // Clear input fields
                $("#phase, #ddlchange_type, #ddlCRF_Complexity, #ddlrelated_work, #development, #changes, #hours, #ddldeveloper, #start_date, #end_date, #costestimation").val("");

                // Show the table and confirm button
                $("#tbl_display").show();

                // Update the sum of the "development" column
                updateDevelopmentSum();

                // Update CodeReview and totalworkhrs values
                updateCodeReviewAndTotalWorkHrs();
            }
        });

        // Add click event handler for remove buttons
        $("#tbtable1").on("click", ".btn-remove", function () {
            $(this).closest("tr").remove();

            // Update the sum of the "development" column after removing a row
            updateDevelopmentSum();

            // Update CodeReview and totalworkhrs values after removing a row
            updateCodeReviewAndTotalWorkHrs();
        });
        const handoverCheckbox = document.getElementById('handoverCheckbox');
        const chooseTechleadDiv = document.getElementById('chooseTechleadDiv');
        handoverCheckbox.addEventListener('change', function () {

            chooseTechleadDiv.style.display = this.checked ? 'block' : 'none';
        });
    });
    function handleSelectChange() {
        var selectElement = document.getElementById('ddlCRF');
         selectedValue = selectElement.value;
        All_Attachments(selectedValue);
    }

    function confirmHoldCRF() {
        var checkbox = document.getElementById("holdCRF");
        if (checkbox.checked) {
            var confirmed = confirm("Are you sure to hold this CRF?");
            if (confirmed) {
                openModal(); 
            } else {
                checkbox.checked = false; 
            }
        }
    }

    function openModal() {
        document.getElementById('crfID').value = selectedValue;
        document.getElementById("crfModal").style.display = "block"; 
    }

    function closeModal() {

        document.getElementById("crfModal").style.display = "none"; 
    }

    // Close the modal when clicking outside of it
    window.onclick = function (event) {
        var modal = document.getElementById("crfModal");
        if (event.target == modal) {
            modal.style.display = "none"; 
        }
    }

</script>
