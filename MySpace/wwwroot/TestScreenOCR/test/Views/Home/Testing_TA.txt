    function calculateHours() {
        var startDate = document.getElementById("start_date").value;
        var endDate = document.getElementById("end_date").value;
        var projectDetails = document.getElementById("project_dtls");
        projectDetails.style.display = (projectDetails.style.display === "none") ? "block" : "none";

        if (startDate && endDate) {
            var startTimestamp = new Date(startDate).getTime();
            var endTimestamp = new Date(endDate).getTime();

            var totalWeekdays = 0;
            var currentDate = new Date(startDate);

            // Loop through each day between start and end date
            while (currentDate <= new Date(endDate)) {
                var dayOfWeek = currentDate.getDay(); // 0 for Sunday, 1 for Monday, etc.
                // Check if the day is not Sunday (i.e., 0)
                if (dayOfWeek !== 0) {
                    totalWeekdays++;
                }
                currentDate.setDate(currentDate.getDate() + 1); // Move to the next day
            }

            var hours = 8 * totalWeekdays;

            var hoursLabel = document.getElementById("hours_label");
            hoursLabel.innerHTML = hours + " Hours left!";
            hoursLabel.style.color = "red";
        } else {
            alert("Please select both start date and end date.");
        }
    }
