<script>

    $(document).ready(function () {
        Bind_CRF_for_usrfeedback();
        $("#loading").hide();
    });
    function setRating(value, id) {
        document.getElementById(id).querySelectorAll(".star").forEach(function (star, index) {
            if (index < value) {
                star.style.color = "gold"; // Set selected stars to gold color
            } else {
                star.style.color = "black"; // Set unselected stars to black color
            }
        });
        document.getElementById(id.replace("ratingStars", "ratingValue")).value = value; // Update hidden input field with the selected value
    }

</script>
